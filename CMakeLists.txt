cmake_minimum_required(VERSION 3.16)
project(kylin-document-scanner VERSION 0.2.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
include(CTest)
option(SCANNER_QT_PKGCONFIG "Use Linux pkg-config Qt metadata (for broken vendor Qt CMake plugins)" OFF)
if(SCANNER_QT_PKGCONFIG)
  if(NOT UNIX OR APPLE)
    message(FATAL_ERROR "SCANNER_QT_PKGCONFIG is intended for Linux builds")
  endif()
  find_package(PkgConfig REQUIRED)
  foreach(component Core Gui Widgets Concurrent PrintSupport)
    pkg_check_modules(QT_${component} REQUIRED IMPORTED_TARGET Qt5${component}>=5.12)
    add_library(Qt5::${component} INTERFACE IMPORTED)
    set_property(TARGET Qt5::${component} PROPERTY INTERFACE_LINK_LIBRARIES PkgConfig::QT_${component})
  endforeach()
  set_property(TARGET Qt5::Core PROPERTY INTERFACE_QT_MAJOR_VERSION 5)
  # Vendor ARM64 Qt uses direct member-function addresses in its metaobject
  # tables. PIC avoids executable PLT addresses breaking typed signal lookup.
  set_property(TARGET Qt5::Core PROPERTY INTERFACE_COMPILE_OPTIONS -fPIC)
  set(Qt5Core_VERSION_MAJOR 5)
  string(REGEX MATCH "^[0-9]+\\.([0-9]+)" qt_version_match "${QT_Core_VERSION}")
  set(Qt5Core_VERSION_MINOR "${CMAKE_MATCH_1}")
  pkg_get_variable(qt_host_bins Qt5Core host_bins)
  find_program(qt_moc NAMES moc HINTS "${qt_host_bins}" /usr/lib/qt5/bin)
  find_program(qt_rcc NAMES rcc HINTS "${qt_host_bins}" /usr/lib/qt5/bin)
  if(NOT qt_moc OR NOT qt_rcc)
    message(FATAL_ERROR "Qt moc/rcc tools are required; install qtbase5-dev-tools")
  endif()
  add_executable(Qt5::moc IMPORTED)
  set_property(TARGET Qt5::moc PROPERTY IMPORTED_LOCATION "${qt_moc}")
  add_executable(Qt5::rcc IMPORTED)
  set_property(TARGET Qt5::rcc PROPERTY IMPORTED_LOCATION "${qt_rcc}")
else()
  find_package(Qt5 5.12 REQUIRED COMPONENTS Core Gui Widgets Concurrent PrintSupport)
endif()
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs videoio)
find_package(Threads REQUIRED)
add_library(scanner_core src/barcode.cpp src/captureframe.cpp src/imageprocessor.cpp src/session.cpp src/exporter.cpp)
target_include_directories(scanner_core PUBLIC src ${OpenCV_INCLUDE_DIRS})
target_link_libraries(scanner_core PUBLIC Qt5::Core Qt5::Gui Qt5::PrintSupport ${OpenCV_LIBS})
add_executable(kylin-document-scanner src/main.cpp src/workstation.cpp src/modes.cpp src/captureshortcut.cpp src/scanstorage.cpp src/mainwindow.cpp src/mainwindow.h src/cropcanvas.cpp src/cropcanvas.h src/camera.cpp src/camera.h resources.qrc)
target_compile_definitions(kylin-document-scanner PRIVATE SCANNER_VERSION="${PROJECT_VERSION}")
target_link_libraries(kylin-document-scanner PRIVATE scanner_core Qt5::Widgets Qt5::Concurrent Threads::Threads)
if(MSVC)
  target_compile_options(scanner_core PRIVATE /utf-8 /W4)
  target_compile_options(kylin-document-scanner PRIVATE /utf-8 /W4)
else()
  target_compile_options(scanner_core PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(kylin-document-scanner PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(GNUInstallDirs)
install(TARGETS kylin-document-scanner RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES packaging/kylin-document-scanner.desktop DESTINATION ${CMAKE_INSTALL_DATAROOTDIR}/applications)
install(FILES packaging/kylin-document-scanner.svg DESTINATION ${CMAKE_INSTALL_DATAROOTDIR}/icons/hicolor/scalable/apps)
install(PROGRAMS scripts/probe-device.sh DESTINATION ${CMAKE_INSTALL_DATAROOTDIR}/kylin-document-scanner)
if(BUILD_TESTING)
  add_executable(test_scanstorage tests/test_scanstorage.cpp src/scanstorage.cpp)
  target_link_libraries(test_scanstorage PRIVATE scanner_core Qt5::Widgets Qt5::Concurrent)
  add_test(NAME scan_storage COMMAND test_scanstorage)
  set_tests_properties(scan_storage PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 45)
  add_executable(test_imageprocessor tests/test_imageprocessor.cpp)
  target_link_libraries(test_imageprocessor PRIVATE scanner_core)
  add_test(NAME image_processing COMMAND test_imageprocessor)
  add_executable(test_barcode tests/test_barcode.cpp)
  target_link_libraries(test_barcode PRIVATE scanner_core)
  add_test(NAME barcode_scanning COMMAND test_barcode)
  add_executable(test_session tests/test_session.cpp)
  target_link_libraries(test_session PRIVATE scanner_core)
  add_test(NAME session_persistence COMMAND test_session)
  add_executable(test_exporter tests/test_exporter.cpp)
  target_link_libraries(test_exporter PRIVATE scanner_core Threads::Threads)
  add_test(NAME export_pipeline COMMAND test_exporter)
  set_tests_properties(export_pipeline PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_executable(test_qt_runtime tests/test_qt_runtime.cpp)
  target_link_libraries(test_qt_runtime PRIVATE Qt5::Core Qt5::Widgets)
  add_test(NAME qt_runtime_signals COMMAND test_qt_runtime)
  set_tests_properties(qt_runtime_signals PROPERTIES TIMEOUT 10 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
if(UNIX AND NOT APPLE AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/packaging/CPackConfig.cmake")
  include(packaging/CPackConfig.cmake)
endif()
